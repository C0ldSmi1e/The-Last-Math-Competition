sha=lambda b:hashlib.sha256(b).hexdigest()
def copy(src,relative):
 target=dest/relative;target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(src,target)
 assert src.read_bytes()==target.read_bytes()
def add_tree(mapping,source,prefix):
 for f in sorted(source.rglob('*')):
  if not f.is_file():continue
  rel=f.relative_to(source)
  if any(x in {'project','.lake','.git','__pycache__'} for x in rel.parts):continue
  if f.suffix in {'.olean','.ilean','.pyc','.o','.a','.dylib','.so'}:continue
  name=str(P(prefix)/rel)
  if f.suffix=='.lean':name+='.txt'
  assert name not in mapping
  mapping[name]=f.read_bytes()
def add_generated(mapping,project,prefix):
 if not project.exists():return
 for f in sorted(project.rglob('*.lean')):
  rel=f.relative_to(project)
  if '.lake' in rel.parts or rel.name in {'Counterexample.lean','Checks.lean'}:continue
  mapping[str(P(prefix)/rel)+'.txt']=f.read_bytes()
def archive(name,mapping):
 rows={key:{'sha256':sha(b),'bytes':len(b)} for key,b in sorted(mapping.items())}
 raw={'files':rows,'file_count':len(rows),'uncompressed_bytes':sum(len(b) for b in mapping.values()),'manifest_excludes_itself':True}
 manifest=(json.dumps(raw,indent=2,sort_keys=True)+'\n').encode()
 mapping={**mapping,'RAW-MANIFEST.json':manifest}
 target=dest/'evidence'/f'{name}.tar.gz';target.parent.mkdir(parents=True,exist_ok=True)
 with target.open('wb') as stream:
  with gzip.GzipFile(filename='',mode='wb',fileobj=stream,mtime=0) as zipped:
   with tarfile.open(fileobj=zipped,mode='w|') as tar:
    for filename,data in sorted(mapping.items()):
     info=tarfile.TarInfo(filename);info.size=len(data);info.mode=0o644;info.mtime=0
     tar.addfile(info,io.BytesIO(data))
 (dest/'evidence'/f'{name}.manifest.json').write_bytes(manifest)
 return {'archive':str(target.relative_to(dest)),'files':len(mapping),'bytes':target.stat().st_size,'sha256':sha(target.read_bytes())}
